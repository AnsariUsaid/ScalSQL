SELECT 
    c.customer_name,
    c.status,
    SUM(o.total_amount) AS total_revenue
FROM 
    customers c
JOIN 
    orders o ON c.id = o.customer_id
WHERE 
    o.order_date >= '2023-10-01'
    AND o.order_date < '2023-11-01'
    AND c.status = 'Active'
GROUP BY 
    c.customer_name, c.status
ORDER BY 
    total_revenue DESC
LIMIT 5;
